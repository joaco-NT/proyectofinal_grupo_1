import { useState, useEffect } from "react";

export const useEntorno = (claseBody = "body-login") => {
  const [esIframe] = useState(() => {
    return typeof window !== "undefined" && window.self !== window.top;
  });

  useEffect(() => {
    if (!esIframe && claseBody) {
      document.body.className = claseBody;
    }

    return () => {
      document.body.className = "";
    };
  }, [esIframe, claseBody]);

  return { esIframe };
};

import { useState, useEffect } from "react";
import { useNavigate } from "react-router-dom";
import { cargarRondasJuego, guardarPuntajeJuego } from "../services/navegador";
import { evaluarRespuesta } from "../services/calculos";

export const useJuego1 = () => {
  const navigate = useNavigate();

  // 1. MEMORIA
  const [rondas, setRondas] = useState([]);
  const [indiceRonda, setIndiceRonda] = useState(0);
  const [puntaje, setPuntaje] = useState(0);

  // 3. EFECTO
  const [retroalimentacion, setRetroalimentacion] = useState({ mensaje: "", claseCSS: "" });
  const [juegoTerminado, setJuegoTerminado] = useState(false);

  useEffect(() => {
    let montado = true;
    cargarRondasJuego("/data/rondasJuego1.json").then((datos) => {
      if (montado) {
        setRondas(datos);
        setIndiceRonda(0);
        setPuntaje(0);
        setJuegoTerminado(false);
        setRetroalimentacion({ mensaje: "", claseCSS: "" });
      }
    });
    return () => {
      montado = false;
    };
  }, []);

  const rondaActual = rondas.length > 0 ? rondas[indiceRonda] : null;

  // 2. ACCIÓN
  const manejarRespuesta = (opcionTexto) => {
    if (retroalimentacion.mensaje !== "") return;

    const resultado = evaluarRespuesta(opcionTexto, rondaActual);
    const nuevoPuntaje = puntaje + resultado.puntosGanados;
    setPuntaje(nuevoPuntaje);

    setRetroalimentacion({
      mensaje: resultado.esCorrecta
        ? "🎉 Correcto. ¡El animal comió!"
        : "❌ Incorrecto. ¡Esa no es su comida!",
      claseCSS: resultado.esCorrecta
        ? "juego1__mensaje--correcto"
        : "juego1__mensaje--incorrecto"
    });

    setTimeout(() => {
      const siguienteIndice = indiceRonda + 1;
      if (siguienteIndice < rondas.length) {
        setIndiceRonda(siguienteIndice);
        setRetroalimentacion({ mensaje: "", claseCSS: "" });
      } else {
        setJuegoTerminado(true);
        guardarPuntajeJuego(1, nuevoPuntaje);
        setRetroalimentacion({ mensaje: "", claseCSS: "" });
      }
    }, 1200);
  };

  const manejarSiguienteJuego = () => {
    navigate("/juego2");
  };

  return {
    memoria: {
      claseBaseEstilo: "juego1",
      emojiUnico: rondaActual?.emoji || "⏳",
      textoAnimal: rondaActual?.animal || "Cargando juego...",
      opciones: rondaActual?.opciones || [],
      puntajeFinal: puntaje
    },
    accion: {
      alElegirOpcion: manejarRespuesta,
      alHacerClickSiguiente: manejarSiguienteJuego
    },
    efecto: {
      juegoTerminado: juegoTerminado,
      estaCargando: rondaActual === null,
      retroalimentacion: retroalimentacion
    }
  };
};

// CUSTOM HOOK: Lógica y Estado de la Música de Fondo
// ==========================================
import { useState, useEffect, useRef } from "react";

export const useMusicaFondo = () => {
  const audioRef = useRef(null);
  const [reproduciendo, setReproduciendo] = useState(false);

  useEffect(() => {
    if (audioRef.current) {
      audioRef.current.volume = 0.10; // Volumen inicial al 10%
    }
  }, []);

  const alternarMusica = async () => {
    const audio = audioRef.current;
    if (!audio) return;

    if (audio.paused) {
      try {
        await audio.play();
        setReproduciendo(true);
        console.log("Música reproduciéndose");
      } catch (error) {
        console.error("Error al reproducir el audio:", error);
        setReproduciendo(false);
      }
    } else {
      audio.pause();
      setReproduciendo(false);
    }
  };

  return {
    audioRef,
    reproduciendo,
    alternarMusica
  };
};

  const [resumen, setResumen] = useState({

    puntajeJuego1: 0,
    puntajeJuego2: 0,

export const filtrarAlumnos = (usuarios) => {
  return usuarios.filter((usuario) => {
    const esAlumnoRol = usuario.rol === "alumno";
    const esAlumnoTipo = usuario.tipo === "alumno";
    return esAlumnoRol || esAlumnoTipo;
  });
};

export const buscarUsuario = (usuarios, identificador) => {
  const textoLimpio = identificador.trim().toLowerCase();
  
  return usuarios.find((usuario) => {
    let coincideEmail = false;
    if (usuario.email !== undefined) {
      coincideEmail = usuario.email.trim().toLowerCase() === textoLimpio;
    }

    let coincideUsuario = false;
    if (usuario.usuario !== undefined) {
      coincideUsuario = usuario.usuario.trim().toLowerCase() === textoLimpio;
    }

    let coincideNombre = false;
    if (usuario.nombre !== undefined) {
      coincideNombre = usuario.nombre.trim().toLowerCase() === textoLimpio;
    }

    return coincideEmail || coincideUsuario || coincideNombre;
  });
};

export const validarAcceso = (usuarios, identificador, password, esIframe) => {
  const textoVacio = !identificador || identificador.trim() === "";
  if (textoVacio === true) {
    return { valido: false, error: "Escribe tu usuario o correo." };
  }

  let usuarioEncontrado = buscarUsuario(usuarios, identificador);

  let esUsuarioNulo = false;
  if (usuarioEncontrado === undefined) {
    esUsuarioNulo = true;
  }

  if (esUsuarioNulo === true && esIframe === true) {
    usuarioEncontrado = {
      nombre: identificador.trim(),
      rol: "alumno",
      juego1: 0,
      juego2: 0,
      esTemporal: true
    };
  }

  let sigueNulo = false;
  if (usuarioEncontrado === undefined) {
    sigueNulo = true;
  }

  if (sigueNulo === true) {
    return { valido: false, error: "Usuario no encontrado." };
  }

  let esAdmin = false;
  if (usuarioEncontrado.rol === "admin") {
    esAdmin = true;
  }

  if (esAdmin === true) {
    let passwordRegistrada = "";
    if (usuarioEncontrado.password !== undefined) {
      passwordRegistrada = String(usuarioEncontrado.password);
    } else if (usuarioEncontrado.contrasena !== undefined) {
      passwordRegistrada = String(usuarioEncontrado.contrasena);
    }

    const passwordEscrita = String(password);
    if (passwordRegistrada !== passwordEscrita) {
      return { valido: false, error: "Contraseña incorrecta." };
    }
  }

  let rutaDestino = "/juego1";
  if (esAdmin === true) {
    rutaDestino = "/admin";
  }

  return {
    valido: true,
    usuario: usuarioEncontrado,
    destino: rutaDestino
  };
};

export const calcularPuntajeTotal = (puntajeJuego1, puntajeJuego2) => {
  const numero1 = Number(puntajeJuego1);
  const numero2 = Number(puntajeJuego2);
  return numero1 + numero2;
};

export const obtenerMensajeFelicitacion = (puntajeTotal, nombreJugador) => {
  const mensajesPorRango = [
    { minimo: 80, texto: "¡Excelente trabajo, " + nombreJugador + "! Dominas los conocimientos." },
    { minimo: 50, texto: "¡Buen trabajo, " + nombreJugador + "! Sigue practicando." },
    { minimo: 0,  texto: "¡A seguir intentándolo, " + nombreJugador + "!" }
  ];

  let mensajeElegido = "¡A seguir intentándolo, " + nombreJugador + "!";

  mensajesPorRango.forEach((rango) => {
    if (puntajeTotal >= rango.minimo) {
      if (mensajeElegido.includes("A seguir")) {
        mensajeElegido = rango.texto;
      }
    }
  });

  return mensajeElegido;
};

export const mezclarOpciones = (opcionesOriginales) => {
  const copiaOpciones = opcionesOriginales.map((opcion) => {
    return opcion;
  });

  return copiaOpciones.sort(() => {
    return Math.random() - 0.5;
  });

// ==========================================
// 1. Sonido de Cursor (Hover) para cualquier botón
// ==========================================
export const reproducirSonidoCurso = () => {
  const tiempoActual = Date.now();

  // Si pasaron menos de 150 milisegundos, ignora para evitar ruidos raros al mover rápido el mouse
  if (tiempoActual - ultimoTiempoAudio < 150) {
    return;
  }

  ultimoTiempoAudio = tiempoActual;
  audioCursoGlobal.pause();         
  audioCursoGlobal.currentTime = 0; 
  audioCursoGlobal.play().catch(() => {});
};

// ==========================================
// 2. Sonido de Clic para cualquier botón
// ==========================================
export const evaluarYReproducirSonido = () => {
  // Reproduce al instante en cualquier botón sin restricciones de texto
  audioClickGlobal.pause();
  audioClickGlobal.currentTime = 0;
  audioClickGlobal.play().catch(() => {
    // Previene errores si el navegador restringe la reproducción automática
  });
};

// CUSTOM HOOK: useLogin.js (Cerebro del Login)
// ==========================================
import { useState } from "react";
import { useNavigate } from "react-router-dom";
import { iniciarSesion } from "../services/navegador";

export const useLogin = (esIframe) => { // 👈 1. Recibe si es iframe desde InicioSesion.jsx
  const navigate = useNavigate();

  // 1. MEMORIA
  const [credenciales, setCredenciales] = useState({
    identificador: "",
    password: ""
  });
  
  // 3. EFECTO
  const [cargando, setCargando] = useState(false);
  const [error, setError] = useState("");

  // 2. ACCIÓN
  const actualizarCampo = (campo) => (e) => {
    const valor = e.target.value;
    setCredenciales((prev) => ({ ...prev, [campo]: valor }));
    if (error) setError(""); 
  };

  const manejarComenzar = async () => {
    // Si no es iframe, exigimos que rellene los campos
    if (!esIframe && (!credenciales.identificador || !credenciales.password)) {
      setError("⚠️ Por favor completa todos los campos.");
      return;
    }

    try {
      setCargando(true);
      setError("");

      // 🚀 2. Enviamos la bandera 'esIframe' a tu iniciarSesion() existente
      const resultado = await iniciarSesion(
        credenciales.identificador, 
        credenciales.password,
        esIframe
      );

      if (resultado.valido === false) {
        setError("❌ Credenciales incorrectas o usuario no encontrado.");
        setCredenciales({ identificador: "", password: "" });
        setCargando(false);
        return;
      }

      navigate(resultado.rutaDestino || "/inicio");

    } catch (err) {
      setError("⚠️ Ocurrió un error al intentar iniciar sesión.");
      setCargando(false);
    }
  };

  return {
    memoria: {
      identificador: credenciales.identificador,
      password: credenciales.password
    },
    accion: {
      alCambiarIdentificador: actualizarCampo("identificador"),
      alCambiarPassword: actualizarCampo("password"),
      alHacerClickIngresar: manejarComenzar
    },
    efecto: {
      error: error,
      cargando: cargando
    }
  };
};

// CUSTOM HOOK: usePanelAdmin.js (Cerebro del Panel Docente)
// ==========================================
import { useState, useEffect, useRef } from "react";
import { useNavigate } from "react-router-dom";
import { obtenerListaAlumnos, cerrarSesion } from "../services/navegador";
import { PANTALLAS_ALUMNO } from "../routes/Rutas";

export const usePanelAdmin = () => {
  const navigate = useNavigate();
  const iframeRef = useRef(null);

  // 1. MEMORIA
  const [pantallaActiva, setPantallaActiva] = useState(1);
  const [listaAlumnos, setListaAlumnos] = useState([]);

  // 3. EFECTO
  const [mostrarTabla, setMostrarTabla] = useState(true);

  useEffect(() => {
    if (mostrarTabla) {
      obtenerListaAlumnos().then((alumnos) => {
        setListaAlumnos(alumnos);
      });
    }
  }, [mostrarTabla]);

  useEffect(() => {
    const manejarMensaje = (event) => {
      if (event.data?.tipo === "CAMBIO_RUTA_IFRAME") {
        const encontrada = PANTALLAS_ALUMNO.find((p) => p.ruta === event.data.pathname);
        if (encontrada) {
          setPantallaActiva(encontrada.numero);
        }
      }
    };

    window.addEventListener("message", manejarMensaje);
    return () => window.removeEventListener("message", manejarMensaje);
  }, []);

  // 2. ACCIÓN
  const irAlMenuDocente = () => {
    setMostrarTabla(true);
  };

  const seleccionarPantalla = (num) => {
    setPantallaActiva(num);
    setMostrarTabla(false);
  };

  const manejarCierreSesion = () => {
    cerrarSesion();
    navigate("/");
  };

  const infoPantalla =
    PANTALLAS_ALUMNO.find((p) => p.numero === pantallaActiva) || PANTALLAS_ALUMNO[0];

  return {
    memoria: {
      pantallaActiva: pantallaActiva,
      listaAlumnos: listaAlumnos,
      iframeRef: iframeRef,
      infoPantalla: infoPantalla
    },
    accion: {
      alIrAlMenuDocente: irAlMenuDocente,
      alSeleccionarPantalla: seleccionarPantalla,
      alCerrarSesion: manejarCierreSesion
    },
    efecto: {
      mostrarTabla: mostrarTabla
    }
  };
};

// LAYOUT RAÍZ / CONTENEDOR GENERAL DE RUTAS

import { useEffect } from "react";
import { Outlet, useLocation } from "react-router-dom";
import "./css/EstiloGlobal.css"; // Estilos base y transiciones transversales
import MusicaFondo from "./components/MusicaFondo"; // 👈 1. Importa el componente de música

export default function App() {
  const ubicacion = useLocation();

  useEffect(() => {
    // Si la aplicación está corriendo dentro de un iframe, notifica al Panel Admin el cambio de ruta en tiempo real
    if (window.self !== window.top) {
      window.parent.postMessage(
        { tipo: "CAMBIO_RUTA_IFRAME", pathname: ubicacion.pathname },
        "*"
      );
    }
  }, [ubicacion.pathname]);

  return (
    <div className="app-contenedor-principal">
      {/* 🎵 2. Renderiza la música de fondo para que esté presente globalmente */}
      <MusicaFondo />

      <div key={ubicacion.pathname} className="transicion-pagina">
        <Outlet />
      </div>
    </div>
  );
}
</file>

## 📁 Componentes (components/)

##  📄  `Boton.jsx`:
Componente de interfaz reutilizable diseñado para encapsular la estructura y el comportamiento de los botones de acción bajo una misma base homogénea.

##  📄  `PlantillaJuegoBase.jsx`:
se implento un componente estructural para unificar la interfaz de todos los minijuegos de la plataforma y evitar duplicar código de diseño entre pantallas.

Lo utilizan directamente las pantallas de juego (como Pantalla02.jsx y Pantalla03.jsx) para renderizar de forma uniforme tanto el estado activo de las preguntas como la pantalla final de puntajes.

## 📁 Estilos (css/)

##  📄  `EstiloGlobal.css`:
 Aplica de forma global para unificar los estilos transversales en todas las pantallas y componentes.

 Al importarse en la raíz, sus reglas se aplican de manera global a todo el proyecto, afectando a todas las pantallas (Pantalla01, Pantalla02, etc.) y componentes sin necesidad de importarlos individualmente en cada archivo.

##  📄  `trasiciones.css`:
Centraliza las reglas de animación, efectos de desvanecimiento y microinteracciones fluidas para los elementos interactivos.

Aporta dinamismo visual y suavidad en la respuesta de los botones, estados de carga y cambios de estado.

Se conectan directamente al archivo raíz o de entrada principal de la aplicación React (app.jsx).

Al importarse en la raíz, sus reglas se aplican de manera global a todo el proyecto, afectando a todas las pantallas (Pantalla01, Pantalla02, etc.) y componentes sin necesidad de importarlos individualmente en cada archivo.

## 📁 Hook (hooks/)

##  📄  `useFondoPantalla.js`:
Añade dinámicamente la clase CSS del cuerpo de la página al montarse y la limpia al desmontarse.

Se conecta directamente al objeto global document.body de cada pantalla para cambiar el fondo exclusivo.

Lo utilizan las pantallas de forma individual (Pantalla01, Pantalla02, etc.) para asignar el color de fondo correspondiente.

##  📄  `useJuego.js`:
Orquesta todo el ciclo de vida, las rondas asíncronas, el puntaje y las respuestas de los minijuegos.

Se conecta con los servicios de cálculo (calculoPantalla02.js y calculoPantalla03.js) y con react-router-dom para la navegación.

Lo utilizan directamente las pantallas de juego (Pantalla02.jsx y Pantalla03.jsx) para encapsular toda su lógica interactiva.

##  📄  `useProgresoPartida.js`:
Proporciona una interfaz limpia y desacoplada para leer y modificar el estado global de la partida.

Se conecta directamente con el módulo de almacenamiento global en memoria (estadoJuego.js).

Lo utilizan las pantallas principales (Pantalla01, Pantalla02, Pantalla03, Pantalla04) para registrar nombres, guardar puntajes y reiniciar el juego.

## 📁 Servicios (services/)

### ⚙️ `estadoJuego.js`:

El módulo que guarda en memoria la sesión actual con el nombre y los puntajes de los minijuegos.
Permite registrar y limitar los puntos mediante funciones seguras sin romper la reactividad.
Borra todo y deja los valores en cero cuando tocamos el botón de reiniciar la partida.

### ⚙️ `utilidades.js`:

Módulo de funciones auxiliares que centraliza la lógica compartida de procesamiento de datos, aleatoriedad y validación para los minijuegos.

Implementa el algoritmo de mezcla Fisher-Yates para el barajado estocástico de opciones y un servicio asíncrono genérico basado en fetch para la ingesta de recursos JSON.

Se importa directamente en calculoPantalla02.js y calculoPantalla03.js para reutilizar las funciones de carga de rondas y validación de respuestas.

Sirve como la base principal que le da soporte a los minijuegos de toda la aplicación sin tener que duplicar lógica.


## 📁 public (data/):

###  `rondasJuego1.json`:
Contiene el conjunto de datos estructurados en formato JSON con los animales, sus emojis y las opciones de comida correspondientes.

Define las rondas del primer minijuego interactivo sobre la alimentación correcta de los animales de la granja.

###  `rondasJuego1.json`:
Almacena las estructuras de datos en formato JSON con los grupos de opciones para identificar al animal intruso.

Define las rondas del segundo minijuego desafiando al usuario a reconocer elementos ajenos al entorno rural.

Se conecta directamente con el servicio calculoPantalla03.js mediante la función encargada de consumir y barajar las rondas.

## 👤 Reporte de Progreso - Miembro 2

### 📒 Datos Generales
- **Estudiante:** Gastón Flores
- **Ejercicio Asignado:**
  - **Ejercicio 03:** Juego 2: El Intruso de la Granja (Manejo de rondas, selección de animales, validación de respuestas y puntaje).
  - **Ejercicio 04:** Pantalla de resumen final (Visualización de puntajes y resultado total de la partida).


### 📄 `Pantalla03.jsx`:
Componente correspondiente al segundo juego de la aplicación, denominado "El Intruso de la Granja". Gestiona el desarrollo de las rondas mediante hooks de React como `useState` y `useEffect`, y utiliza `useNavigate` de `react-router-dom` para controlar la navegación.

El componente presenta diferentes grupos de animales y permite al jugador identificar cuál es el intruso. Al seleccionar una respuesta, valida si es correcta, muestra una retroalimentación visual y actualiza el puntaje. Al finalizar las rondas, guarda el puntaje obtenido mediante `guardarPuntajeJuego2` desde `estadoJuego.js` y redirige a la pantalla de resumen final.

### 📄 `Pantalla04.jsx`:
Componente encargado de mostrar el resumen final de la partida. Recupera los datos almacenados en el servicio `estadoJuego.js` mediante `obtenerResumenPartida()`.

Presenta el nombre del jugador, los puntajes obtenidos en ambos juegos y el puntaje total sobre 100 puntos. También muestra un mensaje de felicitación según el desempeño del jugador y permite reiniciar la partida mediante un botón que limpia los datos guardados y redirige a la pantalla inicial.

---

## 🛠️ Servicios (services/)

### ⚙️ `calculoPantalla03.js`:
Módulo encargado de centralizar la lógica del Juego 2, "El Intruso de la Granja".

- **rondasJuego2:** Arreglo que contiene las rondas del juego, con grupos de animales y la respuesta correcta.
- **calcularPuntajeYValidacion(opcionElegida, opcionCorrecta, puntajeActual):** Evalúa si el animal seleccionado es el intruso, determina si la respuesta es correcta y actualiza el puntaje.
- **prepararGuardadoFinal(puntajeFinal):** Controla el puntaje final para asegurar que no supere los 50 puntos establecidos como máximo para el juego.

### ⚙️ `estadoJuego.js`:
Servicio encargado de almacenar los datos generales de la partida en memoria.

Se utiliza para conservar el nombre del jugador y los puntajes obtenidos durante los juegos.

- **guardarNombreJugador(nombre):** Almacena el nombre del jugador.
- **guardarPuntajeJuego1(puntaje):** Registra el puntaje obtenido en el primer juego.
- **guardarPuntajeJuego2(puntaje):** Registra el puntaje obtenido en el segundo juego.
- **obtenerResumenPartida():** Devuelve una copia de los datos de la partida para que la pantalla de resultados pueda utilizarlos.
- **reiniciarPartida():** Restablece los datos de la partida para comenzar nuevamente.
